// D5 口径：title + "\n" + beforeContent + "\n" + (afterContent ?? "")
export function computeContentHash(
title: string,
beforeContent: string,
afterContent?: string | null,
) {
return createHash("sha256")
.update(`${title}\n${beforeContent}\n${afterContent ?? ""}`, "utf8")
.digest("hex");
}
